(function(){
  var h=$('sbHandle'),sb=$('sidebar'),sx=0,sw=0,moved=false,down=false;
  h.addEventListener('mousedown',function(e){down=true;moved=false;sx=e.clientX;sw=sb.offsetWidth;e.preventDefault();});
  document.addEventListener('mousemove',function(e){if(!down)return;if(Math.abs(e.clientX-sx)>3)moved=true;if(moved){sb.classList.remove('hidden');sb.style.width=Math.max(210,Math.min(430,sw+e.clientX-sx))+'px';}});

})();
