```csharp
// Moto.Core/AI/Embedded/LocalModelContracts.cs
using System.Threading;
using System.Threading.Tasks;

namespace Moto.Core.AI.Embedded;

/// <summary>Requête envoyée à un runtime IA local.</summary>
public sealed record ModelRequest
{
    public string Prompt { get; init; } = "";
    public string SystemPrompt { get; init; } = "";
    public int MaxTokens { get; init; } = 512;
    public double Temperature { get; init; } = 0.2;
}

/// <summary>Réponse d'un runtime IA local (tokens + temps + provenance).</summary>
public sealed record ModelResponse
{
    public bool Success { get; init; }
    public string Text { get; init; } = "";
    public int PromptTokens { get; init; }
    public int CompletionTokens { get; init; }
    public string Provider { get; init; } = "";
    public long ElapsedMs { get; init; }
    public string? Error { get; init; }

    public static ModelResponse Ok(string text, string provider,
        int promptTokens = 0, int completionTokens = 0)
        => new()
        {
            Success = true,
            Text = text,
            Provider = provider,
            PromptTokens = promptTokens,
            CompletionTokens = completionTokens
        };

    public static ModelResponse Fail(string error, string provider = "")
        => new() { Success = false, Error = error, Provider = provider };
}

/// <summary>Abstraction d'un runtime IA local (Ollama, LM Studio, embarqué…).</summary>
public interface ILocalModelRuntime
{
    string Name { get; }
    Task<bool> IsAvailableAsync(CancellationToken ct);
    Task<ModelResponse> GenerateAsync(ModelRequest request, CancellationToken ct);
}
```

```csharp
// Moto.Core/AI/Embedded/OllamaLocalRuntime.cs
using System.Net.Http;
using System.Net.Http.Json;
using System.Text.Json;
using System.Threading;
using System.Threading.Tasks;
using Moto.Core.Settings;

namespace Moto.Core.AI.Embedded;

/// <summary>Runtime local Ollama (HTTP localhost:11434).</summary>
public sealed class OllamaLocalRuntime : ILocalModelRuntime
{
    private static readonly HttpClient Http = new();
    private readonly SettingsEngine _settings;

    public string Name => "ollama";

    public OllamaLocalRuntime(SettingsEngine settings)
    {
        _settings = settings;
    }

    public async Task<bool> IsAvailableAsync(CancellationToken ct)
    {
        try
        {
            var url = _settings.Shared.LocalAi.OllamaUrl.Value;
            var resp = await Http.GetAsync($"{url}/api/tags", ct);
            return resp.IsSuccessStatusCode;
        }
        catch
        {
            return false;
        }
    }

    public async Task<ModelResponse> GenerateAsync(ModelRequest request, CancellationToken ct)
    {
        var url = _settings.Shared.LocalAi.OllamaUrl.Value;
        var model = _settings.Shared.LocalAi.ModelName.Value;

        var payload = new
        {
            model,
            prompt = request.Prompt,
            system = request.SystemPrompt,
            stream = false,
            options = new { num_predict = request.MaxTokens, temperature = request.Temperature }
        };

        var resp = await Http.PostAsJsonAsync($"{url}/api/generate", payload, ct);
        resp.EnsureSuccessStatusCode();

        var json = await resp.Content.ReadAsStringAsync(ct);
        using var doc = JsonDocument.Parse(json);
        var root = doc.RootElement;

        var text = root.TryGetProperty("response", out var r) ? r.GetString() ?? "" : "";
        int promptTokens = root.TryGetProperty("prompt_eval_count", out var p) ? p.GetInt32() : 0;
        int completionTokens = root.TryGetProperty("eval_count", out var c) ? c.GetInt32() : 0;

        return ModelResponse.Ok(text, Name, promptTokens, completionTokens);
    }
}
```

## 📄 `Moto.Core/AI/Embedded/LocalModelService.cs`

```csharp
// Moto.Core/AI/Embedded/LocalModelService.cs
using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Threading;
using System.Threading.Tasks;
using Moto.Core.Logging;
using Moto.Core.Settings;

namespace Moto.Core.AI.Embedded;

/// <summary>
/// MOTO AI Tier 31→32 — Orchestrateur IA local.
/// Pilote un ou plusieurs runtimes locaux avec timeout configurable,
/// fallback automatique et journalisation des tokens/temps.
/// Pure logique : aucune dépendance vers UI / Installer / SignTool.
/// </summary>
public sealed class LocalModelService
{
    private readonly List<ILocalModelRuntime> _runtimes = new();
    private readonly SettingsEngine _settings;
    private readonly StructuredLogCollector _log;
    private readonly object _statsLock = new();

    private int _totalCalls;
    private int _totalPromptTokens;
    private int _totalCompletionTokens;
    private long _totalElapsedMs;

    public LocalModelService(SettingsEngine settings, StructuredLogCollector log)
    {
        _settings = settings;
        _log = log;
    }

    /// <summary>Ajoute un runtime local (ordre = priorité de fallback).</summary>
    public void RegisterRuntime(ILocalModelRuntime runtime) => _runtimes.Add(runtime);

    /// <summary>Génère une réponse en essayant les runtimes dans l'ordre (fallback).</summary>
    public async Task<ModelResponse> GenerateAsync(ModelRequest request, CancellationToken ct = default)
    {
        if (!_settings.Shared.LocalAi.Enabled.Value)
            return ModelResponse.Fail("IA locale désactivée.");

        using var cts = CancellationTokenSource.CreateLinkedTokenSource(ct);
        cts.CancelAfter(TimeSpan.FromSeconds(_settings.Shared.LocalAi.TimeoutSec.Value));

        foreach (var runtime in _runtimes)
        {
            var result = await TryRuntimeAsync(runtime, request, cts.Token);
            if (result.Success) return result;
        }
        return ModelResponse.Fail("Aucun runtime IA local disponible.");
    }

    /// <summary>Statistiques cumulées (appels, tokens, temps).</summary>
    public LocalModelStats GetStats()
    {
        lock (_statsLock)
        {
            return new LocalModelStats
            {
                TotalCalls = _totalCalls,
                TotalPromptTokens = _totalPromptTokens,
                TotalCompletionTokens = _totalCompletionTokens,
                TotalElapsedMs = _totalElapsedMs
            };
        }
    }

    private async Task<ModelResponse> TryRuntimeAsync(
        ILocalModelRuntime runtime, ModelRequest request, CancellationToken ct)
    {
        var sw = Stopwatch.StartNew();
        try
        {
            if (!await runtime.IsAvailableAsync(ct))
                return ModelResponse.Fail("Runtime indisponible.", runtime.Name);

            var response = await runtime.GenerateAsync(request, ct);
            sw.Stop();
            Record(runtime.Name, response, sw.ElapsedMilliseconds);
            return response with { ElapsedMs = sw.ElapsedMilliseconds };
        }
        catch (Exception ex)
        {
            sw.Stop();
            _log.Warning("LocalModelService", $"Échec runtime {runtime.Name}", new { ex.Message });
            return ModelResponse.Fail(ex.Message, runtime.Name);
        }
    }

    private void Record(string provider, ModelResponse response, long elapsedMs)
    {
        lock (_statsLock)
        {
            _totalCalls++;
            _totalPromptTokens += response.PromptTokens;
            _totalCompletionTokens += response.CompletionTokens;
            _totalElapsedMs += elapsedMs;
        }
        _log.Info("LocalModelService", "Appel IA local", new
        {
            provider,
            elapsedMs,
            response.PromptTokens,
            response.CompletionTokens
        });
    }
}

/// <summary>Statistiques cumulées de l'orchestrateur.</summary>
public sealed class LocalModelStats
{
    public int TotalCalls { get; init; }
    public int TotalPromptTokens { get; init; }
    public int TotalCompletionTokens { get; init; }
    public long TotalElapsedMs { get; init; }
}
```

```csharp
// Moto.Editor/Views/LocalAiSettingsView.xaml.cs
using System;
using Moto.Core.AI.Embedded;
using Moto.Core.Settings;

namespace Moto.Editor.Views;

/// <summary>Écran « Paramètres → IA Locale » (liaison fine vers SettingsEngine).</summary>
public partial class LocalAiSettingsView : ContentView
{
    private readonly SettingsEngine _settings;
    private readonly LocalModelService _models;
    private bool _loading;

    public LocalAiSettingsView(LocalModelService models)
    {
        InitializeComponent();
        _settings = SettingsEngine.Shared;
        _models = models;
        Load();
    }

    private void Load()
    {
        _loading = true;
        var ai = _settings.Shared.LocalAi;
        EnabledSwitch.IsToggled = ai.Enabled.Value;
        UrlEntry.Text = ai.OllamaUrl.Value;
        ModelEntry.Text = ai.ModelName.Value;
        TimeoutEntry.Text = ai.TimeoutSec.Value.ToString();
        RefreshStats();
        _loading = false;
    }

    private void OnChanged(object? sender, EventArgs e)
    {
        if (_loading) return;
        var ai = _settings.Shared.LocalAi;
        ai.Enabled.Value = EnabledSwitch.IsToggled;
        ai.OllamaUrl.Value = UrlEntry.Text;
        ai.ModelName.Value = ModelEntry.Text;
        if (int.TryParse(TimeoutEntry.Text, out int t)) ai.TimeoutSec.Value = t;
        RefreshStats();
    }

    private void RefreshStats()
    {
        var s = _models.GetStats();
        StatsLabel.Text = $"Appels : {s.TotalCalls} · Tokens : {s.TotalCompletionTokens} · {s.TotalElapsedMs} ms";
    }
}
```

| Règle | Respectée |
|---|---|
| IA = pure logique (pas d'UI/Installer/SignTool) | ✅ |
| Timeout configurable + fallback automatique | ✅ |
| Journalisation tokens + temps | ✅ (`StructuredLogCollector` + `GetStats`) |

| Méthodes < 40 lignes, XML docs | ✅ |
| Additif, aucun moteur modifié | ✅ |

**Note d'architecture** : le fallback « si le modèle local échoue, Claude Code reprend » est géré **au niveau superviseur** (`MotoAiKernel` / orchestration Tier 32), pas dans `LocalModelService` qui reste déterministe. `LocalModelService` peut être branché comme provider de `MotoAiKernel` sans rien casser.

Prochaine étape possible : **brancher `LocalModelService` dans `MotoAiKernel`** (comme provider prioritaire) ou **`ModelBridge`** (connexion modèles locaux + distants). Lequel voulez-vous ?
