## 📄 `Moto.Core/Performance/PieceTableBuffer.cs`

```csharp
// Moto.Core/Performance/PieceTableBuffer.cs
using System;
using System.Collections.Generic;
using System.Text;

namespace Moto.Core.Performance;

/// <summary>Source d'un morceau de texte dans le PieceTable.</summary>
public enum PieceSource
{
    Original,
    Add
}

/// <summary>
/// Un morceau (piece) référence une portion contiguë dans l'un des deux buffers.
/// </summary>
public readonly struct Piece
{
    public PieceSource Source { get; init; }
    public int Start { get; init; }
    public int Length { get; init; }
}

/// <summary>Snapshot léger de l'état du PieceTable (pour undo/redo).</summary>
internal sealed class PieceTableSnapshot
{
    public List<Piece> Pieces { get; init; } = new();
    public int AddBufferLength { get; init; }
}

/// <summary>
/// Tier 30 — PieceTable : buffer de texte optimisé pour les éditeurs.
/// Stocke le texte original (immuable) + un buffer d'ajouts (append-only) +
/// une table de pieces décrivant l'ordre. Éditions O(1), undo/redo quasi-gratuits.
/// </summary>
public sealed class PieceTableBuffer
{
    private readonly string _originalBuffer;
    private readonly StringBuilder _addBuffer = new();
    private readonly List<Piece> _pieces = new();
    private readonly Stack<PieceTableSnapshot> _undoStack = new();
    private readonly Stack<PieceTableSnapshot> _redoStack = new();
    private readonly int _maxUndoHistory;

    /// <summary>Déclenché quand le contenu change.</summary>
    public event Action? TextChanged;

    /// <summary>Longueur totale du texte.</summary>
    public int Length
    {
        get
        {
            int total = 0;
            foreach (var piece in _pieces) total += piece.Length;
            return total;
        }
    }

    /// <summary>Nombre d'opérations dans l'historique undo.</summary>
    public int UndoCount => _undoStack.Count;

    /// <summary>Nombre d'opérations dans l'historique redo.</summary>
    public int RedoCount => _redoStack.Count;

    public PieceTableBuffer(string initialText, int maxUndoHistory = 1000)
    {
        _originalBuffer = initialText ?? string.Empty;
        _maxUndoHistory = maxUndoHistory;

        if (_originalBuffer.Length > 0)
        {
            _pieces.Add(new Piece
            {
                Source = PieceSource.Original,
                Start = 0,
                Length = _originalBuffer.Length
            });
        }
    }

    /// <summary>Insère du texte à l'offset donné. O(log n) pour trouver la piece, O(1) pour l'ajout.</summary>
    public void Insert(int offset, string text)
    {
        if (string.IsNullOrEmpty(text)) return;
        if (offset < 0 || offset > Length) throw new ArgumentOutOfRangeException(nameof(offset));

        SaveSnapshot();

        int addStart = _addBuffer.Length;
        _addBuffer.Append(text);

        var newPiece = new Piece
        {
            Source = PieceSource.Add,
            Start = addStart,
            Length = text.Length
        };

        if (_pieces.Count == 0 || offset == Length)
        {
            _pieces.Add(newPiece);
        }
        else
        {
            var (pieceIndex, localOffset) = FindPiece(offset);
            if (localOffset == 0)
            {
                _pieces.Insert(pieceIndex, newPiece);
            }
            else
            {
                SplitPiece(pieceIndex, localOffset);
                _pieces.Insert(pieceIndex + 1, newPiece);
            }
        }

        TextChanged?.Invoke();
    }

    /// <summary>Supprime length caractères à partir de offset.</summary>
    public void Delete(int offset, int length)
    {
        if (length <= 0) return;
        if (offset < 0 || offset + length > Length)
            throw new ArgumentOutOfRangeException(nameof(offset));

        SaveSnapshot();

        int remaining = length;
        int currentOffset = offset;

        while (remaining > 0)
        {
            var (pieceIndex, localOffset) = FindPiece(currentOffset);
            var piece = _pieces[pieceIndex];
            int deleteLength = Math.Min(remaining, piece.Length - localOffset);

            if (localOffset == 0 && deleteLength == piece.Length)
            {
                _pieces.RemoveAt(pieceIndex);
            }
            else if (localOffset == 0)
            {
                _pieces[pieceIndex] = new Piece
                {
                    Source = piece.Source,
                    Start = piece.Start + deleteLength,
                    Length = piece.Length - deleteLength
                };
            }
            else if (localOffset + deleteLength == piece.Length)
            {
                _pieces[pieceIndex] = new Piece
                {
                    Source = piece.Source,
                    Start = piece.Start,
                    Length = localOffset
                };
            }
            else
            {
                SplitPiece(pieceIndex, localOffset);
                _pieces.RemoveAt(pieceIndex + 1);
            }

            remaining -= deleteLength;
            currentOffset += deleteLength;
        }

        TextChanged?.Invoke();
    }

    /// <summary>Retourne length caractères à partir de offset.</summary>
    public string GetText(int offset, int length)
    {
        if (length <= 0) return string.Empty;
        if (offset < 0 || offset + length > Length)
            throw new ArgumentOutOfRangeException(nameof(offset));

        var result = new StringBuilder(length);
        int remaining = length;
        int currentOffset = offset;

        while (remaining > 0)
        {
            var (pieceIndex, localOffset) = FindPiece(currentOffset);
            var piece = _pieces[pieceIndex];
            int takeLength = Math.Min(remaining, piece.Length - localOffset);

            var buffer = piece.Source == PieceSource.Original ? _originalBuffer : _addBuffer.ToString();
            result.Append(buffer, piece.Start + localOffset, takeLength);

            remaining -= takeLength;
            currentOffset += takeLength;
        }

        return result.ToString();
    }

    /// <summary>Retourne le texte complet (pour sauvegarde).</summary>
    public string GetFullText() => GetText(0, Length);

    /// <summary>Annule la dernière opération.</summary>
    public bool Undo()
    {
        if (_undoStack.Count == 0) return false;

        _redoStack.Push(CreateSnapshot());
        var snapshot = _undoStack.Pop();
        RestoreSnapshot(snapshot);
        TextChanged?.Invoke();
        return true;
    }

    /// <summary>Refait la dernière opération annulée.</summary>
    public bool Redo()
    {
        if (_redoStack.Count == 0) return false;

        _undoStack.Push(CreateSnapshot());
        var snapshot = _redoStack.Pop();
        RestoreSnapshot(snapshot);
        TextChanged?.Invoke();
        return true;
    }

    /// <summary>Vide l'historique undo/redo.</summary>
    public void ClearHistory()
    {
        _undoStack.Clear();
        _redoStack.Clear();
    }

    private (int PieceIndex, int LocalOffset) FindPiece(int offset)
    {
        int currentOffset = 0;
        for (int i = 0; i < _pieces.Count; i++)
        {
            var piece = _pieces[i];
            if (offset < currentOffset + piece.Length)
            {
                return (i, offset - currentOffset);
            }
            currentOffset += piece.Length;
        }
        return (_pieces.Count, 0);
    }

    private void SplitPiece(int pieceIndex, int localOffset)
    {
        var piece = _pieces[pieceIndex];
        var left = new Piece
        {
            Source = piece.Source,
            Start = piece.Start,
            Length = localOffset
        };
        var right = new Piece
        {
            Source = piece.Source,
            Start = piece.Start + localOffset,
            Length = piece.Length - localOffset
        };

        _pieces[pieceIndex] = left;
        _pieces.Insert(pieceIndex + 1, right);
    }

    private void SaveSnapshot()
    {
        _redoStack.Clear();
        _undoStack.Push(CreateSnapshot());

        if (_undoStack.Count > _maxUndoHistory)
        {
            var temp = new Stack<PieceTableSnapshot>();
            while (_undoStack.Count > _maxUndoHistory) temp.Push(_undoStack.Pop());
            _undoStack.Clear();
            while (temp.Count > 0) _undoStack.Push(temp.Pop());
        }
    }

    private PieceTableSnapshot CreateSnapshot()
    {
        var snapshot = new PieceTableSnapshot
        {
            AddBufferLength = _addBuffer.Length
        };
        snapshot.Pieces.AddRange(_pieces);
        return snapshot;
    }

    private void RestoreSnapshot(PieceTableSnapshot snapshot)
    {
        _pieces.Clear();
        _pieces.AddRange(snapshot.Pieces);
        if (_addBuffer.Length > snapshot.AddBufferLength)
        {
            _addBuffer.Length = snapshot.AddBufferLength;
        }
    }
}
```

## 📄 `Moto.Tests/Performance/PieceTableBufferTests.cs`

```csharp
// Moto.Tests/Performance/PieceTableBufferTests.cs
using Moto.Core.Performance;
using Xunit;

namespace Moto.Tests.Performance;

public class PieceTableBufferTests
{
    [Fact]
    public void Constructor_Initialise_Le_Buffer()
    {
        var buffer = new PieceTableBuffer("Hello");
        Assert.Equal(5, buffer.Length);
        Assert.Equal("Hello", buffer.GetFullText());
    }

    [Fact]
    public void Insert_Ajoute_Au_Milieu()
    {
        var buffer = new PieceTableBuffer("Hello World");
        buffer.Insert(5, " Beautiful");
        Assert.Equal("Hello Beautiful World", buffer.GetFullText());
    }

    [Fact]
    public void Insert_Au_Debut()
    {
        var buffer = new PieceTableBuffer("World");
        buffer.Insert(0, "Hello ");
        Assert.Equal("Hello World", buffer.GetFullText());
    }

    [Fact]
    public void Insert_A_La_Fin()
    {
        var buffer = new PieceTableBuffer("Hello");
        buffer.Insert(5, " World");
        Assert.Equal("Hello World", buffer.GetFullText());
    }

    [Fact]
    public void Delete_Supprime_Au_Milieu()
    {
        var buffer = new PieceTableBuffer("Hello Beautiful World");
        buffer.Delete(5, 10);
        Assert.Equal("Hello World", buffer.GetFullText());
    }

    [Fact]
    public void Delete_Au_Debut()
    {
        var buffer = new PieceTableBuffer("Hello World");
        buffer.Delete(0, 6);
        Assert.Equal("World", buffer.GetFullText());
    }

    [Fact]
    public void Delete_A_La_Fin()
    {
        var buffer = new PieceTableBuffer("Hello World");
        buffer.Delete(5, 6);
        Assert.Equal("Hello", buffer.GetFullText());
    }

    [Fact]
    public void Undo_Annule_Insert()
    {
        var buffer = new PieceTableBuffer("Hello");
        buffer.Insert(5, " World");
        Assert.Equal("Hello World", buffer.GetFullText());

        buffer.Undo();
        Assert.Equal("Hello", buffer.GetFullText());
    }

    [Fact]
    public void Undo_Annule_Delete()
    {
        var buffer = new PieceTableBuffer("Hello World");
        buffer.Delete(5, 6);
        Assert.Equal("Hello", buffer.GetFullText());

        buffer.Undo();
        Assert.Equal("Hello World", buffer.GetFullText());
    }

    [Fact]
    public void Redo_Refait_Operation()
    {
        var buffer = new PieceTableBuffer("Hello");
        buffer.Insert(5, " World");
        buffer.Undo();
        Assert.Equal("Hello", buffer.GetFullText());

        buffer.Redo();
        Assert.Equal("Hello World", buffer.GetFullText());
    }

    [Fact]
    public void Operations_Multiples()
    {
        var buffer = new PieceTableBuffer("AB");
        buffer.Insert(1, "X");  // AXB
        buffer.Insert(2, "Y");  // AXYB
        buffer.Delete(0, 1);    // XYB
        buffer.Insert(0, "Z");  // ZXYB

        Assert.Equal("ZXYB", buffer.GetFullText());
    }

    [Fact]
    public void GetText_Extrait_Une_Partie()
    {
        var buffer = new PieceTableBuffer("Hello World");
        Assert.Equal("World", buffer.GetText(6, 5));
        Assert.Equal("Hello", buffer.GetText(0, 5));
    }

    [Fact]
    public void TextChanged_Est_Declenche()
    {
        var buffer = new PieceTableBuffer("Hello");
        int changeCount = 0;
        buffer.TextChanged += () => changeCount++;

        buffer.Insert(5, " World");
        buffer.Delete(0, 6);
        buffer.Undo();

        Assert.Equal(3, changeCount);
    }

    [Fact]
    public void Grand_Fichier_Performance()
    {
        var largeText = new string('x', 1_000_000);
        var buffer = new PieceTableBuffer(largeText);

        for (int i = 0; i < 100; i++)
        {
            buffer.Insert(500_000, "INSERT");
        }

        Assert.True(buffer.Length > 1_000_000);
    }
}
```
